cat /etc/apache2/sites-enabled/001-kafu-difbac-le-ssl.conf
<IfModule mod_ssl.c>
<VirtualHost *:443>
    ServerName kafu.difbac.com
    ServerAlias www.kafu.difbac.com

    DocumentRoot /var/www/kafu/apps/frontend
    DirectoryIndex index.html index.php

    Alias /api /var/www/fuka/artifacts/kafu-api/public

    <Directory /var/www/fuka/artifacts/kafu-api/public>
        Options FollowSymLinks
        AllowOverride All
        Require all granted
        DirectoryIndex index.php

        <FilesMatch \.php$>
            SetHandler "proxy:unix:/run/php/php8.3-fpm.sock|fcgi://localhost/"
        </FilesMatch>

        RewriteEngine On
        RewriteBase /api/
        RewriteCond %{REQUEST_FILENAME} !-f
        RewriteCond %{REQUEST_FILENAME} !-d
        RewriteRule ^(.*)$ index.php/api/$1 [L,QSA]
    </Directory>

    Alias /cms /var/www/kafu/apps/cms
    <Directory /var/www/kafu/apps/cms>
        Options FollowSymLinks
        AllowOverride All
        Require all granted
        DirectoryIndex index.html

        RewriteEngine On
        RewriteCond %{REQUEST_FILENAME} !-f
        RewriteCond %{REQUEST_FILENAME} !-d
        RewriteRule ^ index.html [L]
    </Directory>

    Alias /staff /var/www/kafu/apps/staff
    <Directory /var/www/kafu/apps/staff>
        Options FollowSymLinks
        AllowOverride All
        Require all granted
        DirectoryIndex index.html

        RewriteEngine On
        RewriteCond %{REQUEST_FILENAME} !-f
        RewriteCond %{REQUEST_FILENAME} !-d
        RewriteRule ^ index.html [L]
    </Directory>

    <Directory /var/www/kafu/apps/frontend>
        Options FollowSymLinks
        AllowOverride All
        Require all granted
        DirectoryIndex index.html

        RewriteEngine On
        RewriteCond %{REQUEST_URI} !^/api
        RewriteCond %{REQUEST_URI} !^/cms
        RewriteCond %{REQUEST_URI} !^/staff
        RewriteCond %{REQUEST_FILENAME} !-f
        RewriteCond %{REQUEST_FILENAME} !-d
        RewriteRule ^ /index.html [L]
    </Directory>

    ErrorLog ${APACHE_LOG_DIR}/kafu-difbac-error.log
    CustomLog ${APACHE_LOG_DIR}/kafu-difbac-access.log combined

Include /etc/letsencrypt/options-ssl-apache.conf
SSLCertificateFile /etc/letsencrypt/live/kafu.difbac.com/fullchain.pem
SSLCertificateKeyFile /etc/letsencrypt/live/kafu.difbac.com/privkey.pem
</VirtualHost>
</IfModule>
